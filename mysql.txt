-- ============================================================
-- Application Access Control Schema
-- Compatible with MySQL 5.5
-- ============================================================

-- ------------------------------------------------------------
-- 1. applications
-- ------------------------------------------------------------
CREATE TABLE applications (
    id          INT             NOT NULL AUTO_INCREMENT,
    name        VARCHAR(100)    NOT NULL,
    slug        VARCHAR(100)    NOT NULL,
    description VARCHAR(255)    NULL,
    url         VARCHAR(255)    NULL,
    is_active   TINYINT(1)      NOT NULL DEFAULT 1,
    created_at  DATETIME        NOT NULL DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT pk_applications PRIMARY KEY (id),
    CONSTRAINT uq_applications_slug UNIQUE (slug)
) ENGINE=InnoDB DEFAULT CHARSET=utf8;

-- ------------------------------------------------------------
-- 2. _groups
-- ------------------------------------------------------------
CREATE TABLE _groups (
    id          INT             NOT NULL AUTO_INCREMENT,
    name        VARCHAR(100)    NOT NULL,
    description VARCHAR(255)    NULL,
    is_active   TINYINT(1)      NOT NULL DEFAULT 1,
    created_at  DATETIME        NOT NULL DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT pk_groups PRIMARY KEY (id),
    CONSTRAINT uq_groups_name UNIQUE (name)
) ENGINE=InnoDB DEFAULT CHARSET=utf8;

-- ------------------------------------------------------------
-- 3. users
-- ------------------------------------------------------------
CREATE TABLE users (
    id            INT             NOT NULL AUTO_INCREMENT,
    username      VARCHAR(50)     NOT NULL,
    email         VARCHAR(150)    NOT NULL,
    password_hash VARCHAR(255)    NOT NULL,
    is_active     TINYINT(1)      NOT NULL DEFAULT 1,
    created_at    DATETIME        NOT NULL DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT pk_users PRIMARY KEY (id),
    CONSTRAINT uq_users_username UNIQUE (username),
    CONSTRAINT uq_users_email UNIQUE (email)
) ENGINE=InnoDB DEFAULT CHARSET=utf8;

-- ------------------------------------------------------------
-- 4. user_groups  (user <-> group junction)
-- ------------------------------------------------------------
CREATE TABLE user_groups (
    user_id     INT         NOT NULL,
    group_id    INT         NOT NULL,
    assigned_at DATETIME    NOT NULL DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT pk_user_groups PRIMARY KEY (user_id, group_id),
    CONSTRAINT fk_ug_user  FOREIGN KEY (user_id)  REFERENCES users(id)  ON DELETE CASCADE ON UPDATE CASCADE,
    CONSTRAINT fk_ug_group FOREIGN KEY (group_id) REFERENCES groups(id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8;

-- ------------------------------------------------------------
-- 5. group_applications  (group <-> application junction)
-- ------------------------------------------------------------
CREATE TABLE group_applications (
    group_id       INT         NOT NULL,
    application_id INT         NOT NULL,
    granted_at     DATETIME    NOT NULL DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT pk_group_applications PRIMARY KEY (group_id, application_id),
    CONSTRAINT fk_ga_group FOREIGN KEY (group_id)       REFERENCES groups(id)       ON DELETE CASCADE ON UPDATE CASCADE,
    CONSTRAINT fk_ga_app   FOREIGN KEY (application_id) REFERENCES applications(id) ON DELETE CASCADE ON UPDATE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8;

-- ------------------------------------------------------------
-- job_settings
-- Single-row configuration table for the email job worker.
-- This table always has exactly one row (id = 1).
-- All reads/writes target that row via WHERE id = 1.
-- ------------------------------------------------------------
CREATE TABLE job_settings (
    id              INT         NOT NULL AUTO_INCREMENT,
    worker_enabled  TINYINT(1)  NOT NULL DEFAULT 1,
    concurrency     INT         NOT NULL DEFAULT 3,
    retry_attempts  INT         NOT NULL DEFAULT 3,
    retry_delay     INT         NOT NULL DEFAULT 60, -- seconds
    updated_at      TIMESTAMP   NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    CONSTRAINT pk_job_settings PRIMARY KEY (id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
